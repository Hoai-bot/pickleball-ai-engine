function renderPassportCard(resData) { document.getElementById('statusBox').style.display = 'none'; document.getElementById('btnSubmit').disabled = false; document.getElementById('passportCard').style.display = 'block'; document.getElementById('resRating').innerText = resData.calculated_rating; document.getElementById('resTier').innerText = resData.tournament_tier; document.getElementById('resElbow').innerText = resData.avg_elbow_angle; document.getElementById('resKnee').innerText = resData.avg_knee_angle; document.getElementById('resComposite').innerText = resData.overall_composite_score; document.getElementById('resPlayerId').innerText = `ID: ${resData.qr_certified_data.player_id}`; if (resData.player_visual_profile) { const prof = resData.player_visual_profile; document.getElementById('resTargetIdentity').innerText = `${prof.court_position} | ${prof.shirt_color} | ${prof.shoe_color} | ${prof.headwear}`; } // 1. RÚT GỌN DỮ LIỆU ĐỂ MÃ QR THƯA VÀ DỄ QUÉT const compactData = { r: resData.calculated_rating, t: resData.tournament_tier, e: resData.avg_elbow_angle, k: resData.avg_knee_angle, c: resData.overall_composite_score, id: resData.qr_certified_data.player_id, s: resData.qr_certified_data.signature }; // 2. MÃ HÓA BASE64 CHUỖI SIÊU NGẮN const encodedPayload = btoa(encodeURIComponent(JSON.stringify(compactData))); const baseUrl = window.location.protocol.startsWith('http') ? `${window.location.origin}${window.location.pathname}` : `${API_BASE_URL}/`; const passportWebUrl = `${baseUrl}?p=${encodedPayload}`; // 3. TẠO ẢNH QR KÍCH THƯỚC LỚN HƠN (250x250) VỚI ĐỘ SẮC NÉT CAO document.getElementById('resQrCode').src = `https://api.qrserver.com/v1/create-qr-code/?size=250x250&data=${encodeURIComponent(passportWebUrl)}`; }
